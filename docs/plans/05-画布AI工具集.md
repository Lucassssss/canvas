> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** 为 AI 对话助手创建与 Joii 画布交互的工具集，让 AI 能够执行创建元素、更新属性、删除元素、获取画布状态等操作，实现通过自然语言控制画布。

**Tech Stack:** Next.js 16, TypeScript, Vercel AI SDK, Zustand (Canvas Store)

```ts
/**
 * 画布工具数据库操作
 */

import { db } from "../../db/index"
import { projects, shapes } from "../../db/schema"
import { eq, and } from "drizzle-orm"

export async function getCanvasStateFromDB(projectId: string) {
  const project = await db.query.projects.findFirst({
    where: eq(projects.id, projectId),
  })

  if (!project) {
    throw new Error("项目不存在")
  }

  const projectShapes = await db.query.shapes.findMany({
    where: eq(shapes.projectId, projectId),
  })

  return {
    projectId,
    viewport: {
      x: project.viewportX || 0,
      y: project.viewportY || 0,
      zoom: project.viewportZoom || 1,
    },
    shapes: projectShapes.map((s) => ({
      id: s.id,
      type: s.type,
      x: s.x,
      y: s.y,
      width: s.width,
      height: s.height,
      rotation: s.rotation || 0,
      opacity: s.opacity || 1,
      ...JSON.parse(s.properties || "{}"),
    })),
    selectedIds: JSON.parse(project.selectedIds || "[]"),
  }
}

export async function createShapeInDB(data: {
  projectId: string
  type: string
  x: number
  y: number
  width: number
  height: number
  [key: string]: any
}) {
  const id = crypto.randomUUID()

  await db.insert(shapes).values({
    id,
    projectId: data.projectId,
    type: data.type,
    x: data.x,
    y: data.y,
    width: data.width,
    height: data.height,
    properties: JSON.stringify(data),
    createdAt: new Date(),
  })

  return id
}

export async function updateShapeInDB(
  projectId: string,
  shapeId: string,
  properties: Record<string, any>
) {
  const shape = await db.query.shapes.findFirst({
    where: and(eq(shapes.projectId, projectId), eq(shapes.id, shapeId)),
  })

  if (!shape) {
    throw new Error("元素不存在")
  }

  const existingProps = JSON.parse(shape.properties || "{}")
  const updatedProps = { ...existingProps, ...properties }

  await db
    .update(shapes)
    .set({
      ...properties,
      properties: JSON.stringify(updatedProps),
      updatedAt: new Date(),
    })
    .where(and(eq(shapes.projectId, projectId), eq(shapes.id, shapeId)))
}

export async function deleteShapeFromDB(projectId: string, shapeId: string) {
  await db
    .delete(shapes)
    .where(and(eq(shapes.projectId, projectId), eq(shapes.id, shapeId)))
}

export async function updateSelectionInDB(
  projectId: string,
  shapeId: string,
  addToSelection: boolean
) {
  const project = await db.query.projects.findFirst({
    where: eq(projects.id, projectId),
  })

  if (!project) {
    throw new Error("项目不存在")
  }

  const currentSelection: string[] = JSON.parse(project.selectedIds || "[]")

  let newSelection: string[]
  if (addToSelection) {
    if (!currentSelection.includes(shapeId)) {
      newSelection = [...currentSelection, shapeId]
    } else {
      newSelection = currentSelection
    }
  } else {
    newSelection = [shapeId]
  }

  await db
    .update(projects)
    .set({
      selectedIds: JSON.stringify(newSelection),
    })
    .where(eq(projects.id, projectId))
}
```

```ts
  projectId: z.string().describe("项目ID"),
  prompt: z.string().describe("图片生成提示词"),

    height: z.number(),

export const generateImageTool = tool({
  description: `根据文字描述生成图片。
适用于需要生成素材、背景、装饰等图片的场景。
生成完成后会返回图片URL，可以直接插入到画布上。`,
  inputSchema: generateImageSchema,
  execute: async ({ userId, projectId, prompt, model, resolution }) => {
    try {
      // 检查积分
      const user = await db.query.users.findFirst({
        where: eq(users.id, userId),
      })

      if (!user) {
        return { success: false, error: "用户不存在" }
      }

      const cost = 1 // 默认消耗积分
      if (user.credits < cost) {
        return { success: false, error: "积分不足" }
      }

      // 生成图片
      const result = await imageGenerationService.generate({
        combinationTypeId: "simple-tryon",
        slotContents: {},
        settings: {
          prompt,
          model: model || "gemini-2.5-flash-image",
          resolution: resolution || { width: 1024, height: 1024 },
        },
      })

      if (!result.success) {
        return { success: false, error: result.error }
      }

      // 扣除积分
      await creditService.consume(userId, cost, `图片生成: ${prompt.substring(0, 50)}`)

      // 在画布上创建图片元素
      const shapeId = await createShapeInDB({
        projectId,
        type: "image",
        x: 100,
        y: 100,
        width: 400,
        height: 400,
        imageUrl: result.imageUrl,
      })

        success: true,
        imageUrl: result.imageUrl,
        shapeId,
        message: `成功生成图片并添加到画布`,
```

## Task 7: 创建前端画布状态同步

**Files:**
- Create: `apps/web/src/features/canvas-sync/CanvasSyncProvider.tsx`

**Step 1: 创建画布状态同步 Provider**

```tsx
'use client'

import React, { useEffect, useCallback, useRef } from 'react'
import { useCanvasStore } from '@/app/canvas/store'

interface CanvasSyncProviderProps {
  projectId: string
  children: React.ReactNode
}

export function CanvasSyncProvider({ projectId, children }: CanvasSyncProviderProps) {
  const { shapes, selectedIds, viewport } = useCanvasStore()
  const lastSyncRef = useRef<string>('')

  // 同步画布状态到服务器
  const syncState = useCallback(async () => {
    const stateKey = JSON.stringify({ shapes, selectedIds, viewport })
    if (stateKey === lastSyncRef.current) return

    try {
      await fetch(`/api/projects/${projectId}/canvas-state`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        credentials: 'include',
        body: JSON.stringify({ shapes, selectedIds, viewport }),
      })
      lastSyncRef.current = stateKey
    } catch (error) {
      console.error('Failed to sync canvas state:', error)
    }
  }, [projectId, shapes, selectedIds, viewport])

  // 定期同步
  useEffect(() => {
    const interval = setInterval(syncState, 5000) // 每5秒同步一次
    return () => clearInterval(interval)
  }, [syncState])

  // 重要操作后立即同步
  const shapesRef = useRef(shapes)
  shapesRef.current = shapes

  useEffect(() => {
    if (shapes.length !== shapesRef.current.length) {
      syncState()
    }
  }, [shapes.length, syncState])

  return <>{children}</>
}
```
