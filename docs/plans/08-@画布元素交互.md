# @画布元素 交互开发计划

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** 实现用户在 AI 对话中通过 `@元素ID` 或 `@选择` 语法引用画布上的元素，让 AI 能够基于引用元素进行理解和操作。

**Architecture:** 前端解析用户输入中的 `@` 引用语法，将引用的元素信息作为上下文注入到 AI 对话中。AI 能够理解引用并基于引用元素执行操作（如"参考这个模特的姿势"、"把这件衣服换到那个模特身上"）。

**Tech Stack:** Next.js 16, React, TypeScript, 正则表达式

## Task 1: 创建 @引用解析器

**Step 1: 创建 @mention 解析函数**

```ts
/**
 * @mention 解析器
 *
 * 解析用户消息中的 @ 引用语法：
 * - @element-name - 按名称引用元素
 * - @selected - 引用当前选中的元素
 * - @last - 引用最近添加的元素
 */

export interface MentionRef {
  type: 'id' | 'name' | 'selected' | 'last' | 'all'
  value?: string
  elementIds?: string[]
}

export interface ParsedMessage {
  text: string
  mentions: MentionRef[]
}

const MENTION_PATTERNS = [
  // @selected - 当前选中的元素
  {
    regex: /@selected/gi,
    type: 'selected' as const,
  },
  // @last - 最近添加的元素
  {
    regex: /@last/gi,
    type: 'last' as const,
  },
  // @all - 所有元素
  {
    regex: /@all/gi,
    type: 'all' as const,
  },
  // @元素ID 或 @元素名称
  {
    regex: /@([a-zA-Z0-9_-]+)/g,
    type: 'name' as const,
  },
]

export function parseMentions(
  message: string,
  canvasState: {
    shapes: Array<{
      id: string
      type: string
      name?: string
      imageUrl?: string
      text?: string
    }>
    selectedIds: string[]
  }
): ParsedMessage {
  const mentions: MentionRef[] = []
  let processedMessage = message

  for (const pattern of MENTION_PATTERNS) {
    const matches = message.matchAll(pattern.regex)
    for (const match of matches) {
      const fullMatch = match[0]
      const value = match[1]

      if (pattern.type === 'selected') {
        mentions.push({
          type: 'selected',
          elementIds: canvasState.selectedIds,
        })
      } else if (pattern.type === 'last') {
        const lastShape = canvasState.shapes[canvasState.shapes.length - 1]
        if (lastShape) {
          mentions.push({
            type: 'last',
            elementIds: [lastShape.id],
          })
        }
      } else if (pattern.type === 'all') {
        mentions.push({
          type: 'all',
          elementIds: canvasState.shapes.map((s) => s.id),
        })
      } else if (pattern.type === 'name') {
        // 查找匹配的元素
        const matchedShape = canvasState.shapes.find(
          (s) =>
            s.id === value ||
            s.name === value ||
            s.type === value ||
            (s.text && s.text.includes(value))
        )
        if (matchedShape) {
          mentions.push({
            type: 'id',
            value,
            elementIds: [matchedShape.id],
          })
        }
      }

      // 将 @mention 替换为可读的引用标记
      processedMessage = processedMessage.replace(
        fullMatch,
        `[${fullMatch.substring(1)}]`
      )
    }
  }

  return {
    text: processedMessage,
    mentions,
  }
}
```

```ts
/**
 * 元素引用上下文构建器
 *
 * 将引用的元素转换为 AI 可理解的上下文描述
 */

import { ShapeProps } from '@/app/canvas/shapes/types'

export interface ElementContext {
  elementId: string
  description: string
  imageUrl?: string
  attributes: Record<string, any>
}

export function buildElementContext(shapes: ShapeProps[]): ElementContext[] {
  return shapes.map((shape) => {
    const baseContext: ElementContext = {
      elementId: shape.id,
      description: getElementDescription(shape),
      attributes: {
        type: shape.type,
        position: { x: shape.x, y: shape.y },
        size: { width: shape.width, height: shape.height },
        rotation: shape.rotation,
      },
    }

    // 添加类型特定的属性
    if (shape.type === 'image' && shape.imageUrl) {
      baseContext.imageUrl = shape.imageUrl
      baseContext.description = `图片元素：${shape.imageName || '未命名图片'}，尺寸 ${shape.imageWidth}x${shape.imageHeight}`
    }

    if (shape.type === 'ai-combination') {
      baseContext.description = `AI 换装组件，类型：${shape.combinationTypeId}`
      baseContext.attributes.slotContents = shape.slotContents
    }

    if (shape.type === 'text' && shape.text) {
      baseContext.description = `文本元素：${shape.text}`
    }

    return baseContext
  })
}

function getElementDescription(shape: ShapeProps): string {
  switch (shape.type) {
    case 'image':
      return shape.imageName || '图片'
    case 'text':
      return shape.text?.substring(0, 50) || '文本'
    case 'ai-combination':
      return `AI 换装组件`
    case 'rect':
      return '矩形'
    case 'circle':
      return '圆形'
    default:
      return shape.type
  }
}

export function formatContextForAI(
  elementContexts: ElementContext[],
  mentionType: 'selected' | 'last' | 'all' | 'id'
): string {
  if (elementContexts.length === 0) {
    return '（引用的元素不存在）'
  }

  const descriptions = elementContexts.map((ctx, index) => {
    let text = `[${index + 1}] ${ctx.description}`
    if (ctx.imageUrl) {
      text += `\n   图片URL: ${ctx.imageUrl}`
    }
    Object.entries(ctx.attributes).forEach(([key, value]) => {
      if (key !== 'imageUrl' && typeof value === 'object') {
        text += `\n   ${key}: ${JSON.stringify(value)}`
      }
    })
    return text
  })

  const prefixMap = {
    selected: '当前选中的元素：',
    last: '最近添加的元素：',
    all: '画布上的所有元素：',
    id: '引用的元素：',
  }

  return `${prefixMap[mentionType]}\n${descriptions.join('\n')}`
}
```

```tsx
import { useState, useRef, useCallback, useEffect } from 'react'
import { parseMentions, ParsedMessage } from '../utils/mentionParser'
import { useCanvasStore } from '@/app/canvas/store'

export function ChatInput({
  onSend,
}: {
  onSend: (message: string, context: any) => void
}) {
  const [input, setInput] = useState('')
  const [showMentionPicker, setShowMentionPicker] = useState(false)
  const [mentionSearch, setMentionSearch] = useState('')
  const inputRef = useRef<HTMLTextAreaElement>(null)
  const { shapes, selectedIds } = useCanvasStore()

  const handleInputChange = (e: React.ChangeEvent<HTMLTextAreaElement>) => {
    const value = e.target.value
    setInput(value)

    // 检测 @ 触发
    const cursorPos = e.target.selectionStart
    const textBeforeCursor = value.substring(0, cursorPos)
    const lastAtIndex = textBeforeCursor.lastIndexOf('@')

    if (lastAtIndex !== -1 && lastAtIndex === cursorPos - 1) {
      setShowMentionPicker(true)
      setMentionSearch('')
    } else if (showMentionPicker) {
      // 检查是否还在 @ 状态
      const textAfterAt = textBeforeCursor.substring(lastAtIndex + 1)
      if (textAfterAt.includes(' ') || textAfterAt.includes('\n')) {
        setShowMentionPicker(false)
      } else {
        setMentionSearch(textAfterAt)
      }
    }
  }

  const insertMention = (mention: string) => {
    const cursorPos = inputRef.current?.selectionStart || 0
    const textBeforeCursor = input.substring(0, cursorPos)
    const lastAtIndex = textBeforeCursor.lastIndexOf('@')

    const newText =
      input.substring(0, lastAtIndex) +
      `@${mention} ` +
      input.substring(cursorPos)

    setInput(newText)
    setShowMentionPicker(false)
    inputRef.current?.focus()
  }

  const handleSend = () => {
    if (!input.trim()) return

    // 解析消息中的 @ 引用
    const parsed: ParsedMessage = parseMentions(input, { shapes, selectedIds })

    // 构建上下文
    const context = {
      mentions: parsed.mentions,
      rawMessage: input,
    }

    onSend(parsed.text, context)
    setInput('')
  }

  // 过滤可引用的元素
  const filteredShapes = shapes.filter((s) => {
    if (!mentionSearch) return true
    const search = mentionSearch.toLowerCase()
    return (
      s.id.toLowerCase().includes(search) ||
      s.type.toLowerCase().includes(search) ||
      (s as any).name?.toLowerCase().includes(search) ||
      (s as any).text?.toLowerCase().includes(search)
    )
  })

  return (
    <div className="relative">
      {/* @ 引用选择器 */}
      {showMentionPicker && (
        <div className="absolute bottom-full left-0 mb-2 w-64 bg-white rounded-lg shadow-lg border max-h-64 overflow-y-auto">
          <div className="p-2 border-b text-xs text-gray-500">
            选择引用的元素
          </div>

          {selectedIds.length > 0 && (
            <button
              onClick={() => insertMention('selected')}
              className="w-full px-3 py-2 text-left hover:bg-gray-50 text-sm"
            >
              <span className="text-blue-500">@selected</span> - 当前选中 (
              {selectedIds.length})
            </button>
          )}

          <button
            onClick={() => insertMention('last')}
            className="w-full px-3 py-2 text-left hover:bg-gray-50 text-sm"
          >
            <span className="text-blue-500">@last</span> - 最近添加
          </button>

          {filteredShapes.map((shape) => (
            <button
              key={shape.id}
              onClick={() => insertMention(shape.id)}
              className="w-full px-3 py-2 text-left hover:bg-gray-50 text-sm flex items-center gap-2"
            >
              {shape.type === 'image' && shape.imageUrl && (
                <img
                  src={shape.imageUrl}
                  className="w-8 h-8 object-cover rounded"
                />
              )}
              <span>
                <span className="text-blue-500">@{shape.id.substring(0, 8)}</span>
                <span className="text-gray-400 ml-1">{shape.type}</span>
              </span>
            </button>
          ))}
        </div>
      )}

      {/* 输入框 */}
      <div className="flex gap-2">
        <textarea
          ref={inputRef}
          value={input}
          onChange={handleInputChange}
          placeholder="输入消息... 使用 @ 引用画布元素"
          className="flex-1 px-4 py-3 border rounded-xl resize-none focus:outline-none focus:ring-2 focus:ring-blue-500"
          rows={2}
          onKeyDown={(e) => {
            if (e.key === 'Enter' && !e.shiftKey) {
              e.preventDefault()
              handleSend()
            }
          }}
        />
        <button
          onClick={handleSend}
          className="px-6 py-3 bg-blue-500 text-white rounded-xl hover:bg-blue-600"
        >
          发送
        </button>
      </div>
    </div>
  )
}
```

```ts
interface ChatRequest {
  message: string
  projectId: string
  mentionContext?: {
    mentions: Array<{
      type: 'selected' | 'last' | 'all' | 'id'
      elementIds?: string[]
    }>
  }
}

router.post('/chat', async (req: Request, res: Response) => {
  const { message, projectId, mentionContext } = req.body as ChatRequest

  // 处理 @ 引用上下文
  let elementContexts: any[] = []
  if (mentionContext?.mentions) {
    elementContexts = await getElementContextsFromMentions(
      projectId,
      mentionContext.mentions
    )
  }

  // 将元素上下文注入到消息中
  const enhancedMessage = elementContexts.length > 0
    ? `${message}\n\n[画布元素上下文]\n${JSON.stringify(elementContexts, null, 2)}`
    : message

  const result = await agentService.chat(
    req.user!.id,
    projectId,
    enhancedMessage
  )

  res.json(result)
})

async function getElementContextsFromMentions(
  projectId: string,
  mentions: ChatRequest['mentionContext']['mentions']
) {
  const contexts = []

  for (const mention of mentions) {
    if (mention.elementIds) {
      for (const elementId of mention.elementIds) {
        const shape = await getShapeById(projectId, elementId)
        if (shape) {
          contexts.push({
            type: mention.type,
            elementId: shape.id,
            elementType: shape.type,
            imageUrl: shape.imageUrl,
            text: shape.text,
            attributes: JSON.parse(shape.properties || '{}'),
          })
        }
      }
    }
  }

  return contexts
}
```

```markdown
## @ 元素引用

用户可以在消息中使用 @ 语法引用画布上的元素：

- `@selected` - 引用当前选中的元素
- `@last` - 引用最近添加的元素
- `@all` - 引用画布上的所有元素
- `@元素ID` - 引用特定元素

当用户引用元素时，你会收到该元素的详细信息，包括：
- 元素类型（图片、文本、AI组件等）
- 元素属性（位置、尺寸、旋转等）
- 如果是图片，还会提供图片URL

**示例理解能力：**
- "把这个模特的姿势换成那个" → 理解两个被引用元素的图片
- "在这件衣服上添加图案" → 理解被引用的是服装图
- "将选中的图片右移 100px" → 理解 selected 引用的元素
```

```tsx
'use client'

import React, { useState } from 'react'
import { Search, Image, Type, Box, Sparkles } from 'lucide-react'

interface ElementMentionPickerProps {
  shapes: Array<{
    id: string
    type: string
    imageUrl?: string
    text?: string
    name?: string
  }>
  selectedIds: string[]
  onSelect: (shapeId: string) => void
  onSelectSelected: () => void
  onSelectLast: () => void
}

export function ElementMentionPicker({
  shapes,
  selectedIds,
  selectedIds: [],
  onSelect,
  onSelectSelected,
  onSelectLast,
}: ElementMentionPickerProps) {
  const [search, setSearch] = useState('')

  const filteredShapes = shapes.filter((s) => {
    if (!search) return true
    const q = search.toLowerCase()
    return (
      s.id.toLowerCase().includes(q) ||
      s.type.toLowerCase().includes(q) ||
      s.name?.toLowerCase().includes(q) ||
      s.text?.toLowerCase().includes(q)
    )
  })

  const getIcon = (type: string) => {
    switch (type) {
      case 'image':
        return <Image className="w-4 h-4" />
      case 'text':
        return <Type className="w-4 h-4" />
      case 'ai-combination':
        return <Sparkles className="w-4 h-4" />
      default:
        return <Box className="w-4 h-4" />
    }
  }

  return (
    <div className="bg-white rounded-xl shadow-xl border w-72 max-h-80 overflow-hidden">
      <div className="p-2 border-b">
        <div className="relative">
          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
          <input
            type="text"
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            placeholder="搜索元素..."
            className="w-full pl-9 pr-3 py-2 text-sm border rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500"
          />
        </div>
      </div>

      <div className="p-2 space-y-1 max-h-60 overflow-y-auto">
        {selectedIds.length > 0 && (
          <button
            onClick={onSelectSelected}
            className="w-full px-3 py-2 text-left hover:bg-blue-50 rounded-lg flex items-center gap-3"
          >
            <div className="w-8 h-8 bg-blue-100 rounded flex items-center justify-center">
              <span className="text-blue-600 text-xs font-medium">
                {selectedIds.length}
              </span>
            </div>
            <div>
              <div className="text-sm font-medium text-blue-600">@selected</div>
              <div className="text-xs text-gray-500">
                当前选中 ({selectedIds.length} 个)
              </div>
            </div>
          </button>
        )}

        <button
          onClick={onSelectLast}
          className="w-full px-3 py-2 text-left hover:bg-gray-50 rounded-lg flex items-center gap-3"
        >
          <div className="w-8 h-8 bg-gray-100 rounded flex items-center justify-center">
            <span className="text-gray-600 text-xs">↑1</span>
          </div>
          <div>
            <div className="text-sm font-medium">@last</div>
            <div className="text-xs text-gray-500">最近添加</div>
          </div>
        </button>

        {filteredShapes.map((shape) => (
          <button
            key={shape.id}
            onClick={() => onSelect(shape.id)}
            className="w-full px-3 py-2 text-left hover:bg-gray-50 rounded-lg flex items-center gap-3"
          >
            {shape.type === 'image' && shape.imageUrl ? (
              <img
                src={shape.imageUrl}
                className="w-8 h-8 object-cover rounded"
              />
            ) : (
              <div className="w-8 h-8 bg-gray-100 rounded flex items-center justify-center">
                {getIcon(shape.type)}
              </div>
            )}
            <div className="flex-1 min-w-0">
              <div className="text-sm font-medium truncate">
                {shape.name || shape.id.substring(0, 8)}
              </div>
              <div className="text-xs text-gray-500">{shape.type}</div>
            </div>
          </button>
        ))}
      </div>
    </div>
  )
}
```
