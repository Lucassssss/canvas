```ts
/**
 * Kling (快手可灵) 视频生成服务商适配器
 */

import { VideoGenerationOptions, VideoGenerateResult, VideoStatus } from "../types"

const KLING_API_URL = process.env.KLING_API_URL || "https://api.kling.ai/v1"
const KLING_API_KEY = process.env.KLING_API_KEY

export class KlingProvider {
  readonly id = "kling"
  readonly name = "Kling (快手可灵)"
  readonly supportedModes = ["kling-standard", "kling-hd"]

  async generate(options: VideoGenerationOptions): Promise<VideoGenerateResult> {
    if (!KLING_API_KEY) {
      return { success: false, error: "Kling API Key 未配置" }
    }

    try {
      const { model, prompt, imageUrl, aspectRatio, duration } = options

      const requestBody: any = {
        model: model === "kling-hd" ? "kling_hd" : "kling_v1",
        prompt,
        aspect_ratio: aspectRatio,
        duration: duration === 10 ? "10s" : "5s",
      }

      if (imageUrl) {
        requestBody.image_url = imageUrl
      }

      const response = await fetch(`${KLING_API_URL}/videos/generations`, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "Authorization": `Bearer ${KLING_API_KEY}`,
        },
        body: JSON.stringify(requestBody),
      })

      if (!response.ok) {
        const error = await response.text()
        return { success: false, error: `Kling API 错误: ${response.status}` }
      }

      const data = await response.json()
      return {
        success: true,
        taskId: data.task_id,
        status: VideoStatus.PENDING,
      }
    } catch (error) {
      return {
        success: false,
        error: error instanceof Error ? error.message : "视频生成失败",
      }
    }
  }

  async queryStatus(taskId: string): Promise<VideoGenerateResult> {
    if (!KLING_API_KEY) {
      return { success: false, error: "Kling API Key 未配置" }
    }

    try {
      const response = await fetch(`${KLING_API_URL}/videos/generations/${taskId}`, {
        headers: {
          "Authorization": `Bearer ${KLING_API_KEY}`,
        },
      })

      if (!response.ok) {
        return { success: false, error: `查询失败: ${response.status}` }
      }

      const data = await response.json()

      const statusMap: Record<string, VideoStatus> = {
        "pending": VideoStatus.PENDING,
        "processing": VideoStatus.PROCESSING,
        "completed": VideoStatus.COMPLETED,
        "failed": VideoStatus.FAILED,
      }

      return {
        success: true,
        taskId,
        status: statusMap[data.status] || VideoStatus.PROCESSING,
        videoUrl: data.video_url,
      }
    } catch (error) {
      return {
        success: false,
        error: error instanceof Error ? error.message : "查询失败",
      }
    }
  }
}
```

```ts
/**
 * 视频生成服务
 */

import { KlingProvider } from "./providers/kling"
import {
  VideoGenerateInput,
  VideoGenerateResult,
  VideoTask,
  VideoModel,
  VideoStatus,
} from "./types"
import { db } from "../../db/index"
import { creditTransactions, users } from "../../db/schema"
import { eq } from "drizzle-orm"

const CREDIT_COSTS: Record<VideoModel, number> = {
  [VideoModel.KLING_STANDARD]: 10,
  [VideoModel.KLING_HD]: 20,
  [VideoModel.RUNWAY_GEN3]: 15,
  [VideoModel.PIKA_2]: 10,
}

export class VideoService {
  private providers: Map<VideoModel, any> = new Map()

  constructor() {
    this.providers.set(VideoModel.KLING_STANDARD, new KlingProvider())
    this.providers.set(VideoModel.KLING_HD, new KlingProvider())
  }

  async generate(
    userId: string,
    input: VideoGenerateInput
  ): Promise<VideoGenerateResult> {
    const provider = this.providers.get(input.model)
    if (!provider) {
      return { success: false, error: `不支持的模型: ${input.model}` }
    }

    // 检查用户积分
    const user = await db.query.users.findFirst({
      where: eq(users.id, userId),
    })

    const cost = CREDIT_COSTS[input.model]
    if (!user || user.credits < cost) {
      return { success: false, error: "积分不足" }
    }

    // 调用服务商生成
    const result = await provider.generate({
      model: input.model,
      prompt: input.prompt,
      imageUrl: input.imageUrl,
      aspectRatio: input.aspectRatio,
      duration: input.duration,
      seed: input.seed,
    })

    if (!result.success) {
      return result
    }

    // 扣除积分（先冻结，生成完成后确认）
    await db.update(users)
      .set({ credits: user.credits - cost })
      .where(eq(users.id, userId))

    // 记录任务到数据库
    await db.insert(videoTasks).values({
      taskId: result.taskId,
      userId,
      status: VideoStatus.PENDING,
      model: input.model,
      prompt: input.prompt,
      imageUrl: input.imageUrl,
      aspectRatio: input.aspectRatio,
      duration: input.duration,
      creditsCost: cost,
      createdAt: new Date(),
    })

    return result
  }

  async queryStatus(taskId: string): Promise<VideoGenerateResult> {
    const task = await db.query.videoTasks.findFirst({
      where: eq(videoTasks.taskId, taskId),
    })

    if (!task) {
      return { success: false, error: "任务不存在" }
    }

    const provider = this.providers.get(task.model as VideoModel)
    if (!provider) {
      return { success: false, error: "服务商不存在" }
    }

    const result = await provider.queryStatus(taskId)

    // 更新任务状态
    if (result.status !== task.status) {
      await db.update(videoTasks)
        .set({
          status: result.status,
          videoUrl: result.videoUrl,
          completedAt: result.status === VideoStatus.COMPLETED ? new Date() : undefined,
        })
        .where(eq(videoTasks.taskId, taskId))

      // 如果失败，退还积分
      if (result.status === VideoStatus.FAILED) {
        const user = await db.query.users.findFirst({
          where: eq(users.id, task.userId),
        })
        if (user) {
          await db.update(users)
            .set({ credits: user.credits + task.creditsCost })
            .where(eq(users.id, task.userId))
        }
      }
    }

    return result
  }

  async getUserTasks(userId: string): Promise<VideoTask[]> {
    const tasks = await db.query.videoTasks.findMany({
      where: eq(videoTasks.userId, userId),
      orderBy: (tasks, { desc }) => [desc(tasks.createdAt)],
    })
    return tasks
  }
}

export const videoService = new VideoService()
```

```ts
import { Router, Request, Response } from "express"
import { requireAuth } from "../middleware/auth"
import { videoService } from "../services/video"
import { VideoModel, VideoAspectRatio, VideoDuration } from "../services/video/types"

const router = Router()

router.use(requireAuth)

router.post("/generate", async (req: Request, res: Response) => {
  try {
    const userId = req.user!.id
    const { model, prompt, imageUrl, aspectRatio, duration } = req.body

    if (!model || !prompt) {
      return res.status(400).json({ success: false, error: "缺少必要参数" })
    }

    const result = await videoService.generate(userId, {
      model,
      prompt,
      imageUrl,
      aspectRatio: aspectRatio || VideoAspectRatio.RATIO_16_9,
      duration: duration || VideoDuration.STANDARD,
    })

    res.json(result)
  } catch (error) {
    console.error("视频生成失败:", error)
    res.status(500).json({ success: false, error: "服务器错误" })
  }
})

router.get("/task/:taskId", async (req: Request, res: Response) => {
  try {
    const { taskId } = req.params
    const result = await videoService.queryStatus(taskId)
    res.json(result)
  } catch (error) {
    res.status(500).json({ success: false, error: "服务器错误" })
  }
})

router.get("/tasks", async (req: Request, res: Response) => {
  try {
    const userId = req.user!.id
    const tasks = await videoService.getUserTasks(userId)
    res.json({ success: true, tasks })
  } catch (error) {
    res.status(500).json({ success: false, error: "服务器错误" })
  }
})

export { router as videoRouter }
```

```tsx
'use client'

import React, { useState, useCallback } from 'react'
import { Play, Upload, Download, Loader2 } from 'lucide-react'
import { useAuth } from '@/features/auth/useAuth'
import { useCredits } from '@/features/credits/useCredits'

interface VideoGeneratorProps {
  initialImageUrl?: string
  onVideoGenerated?: (videoUrl: string) => void
}

const VIDEO_MODELS = [
  { id: 'kling-standard', name: 'Kling 标准版', cost: 10 },
  { id: 'kling-hd', name: 'Kling HD', cost: 20 },
]

const ASPECT_RATIOS = [
  { id: '16:9', name: '横屏 16:9' },
  { id: '9:16', name: '竖屏 9:16' },
  { id: '1:1', name: '方形 1:1' },
]

export function VideoGenerator({ initialImageUrl, onVideoGenerated }: VideoGeneratorProps) {
  const [model, setModel] = useState('kling-standard')
  const [prompt, setPrompt] = useState('')
  const [imageUrl, setImageUrl] = useState(initialImageUrl)
  const [aspectRatio, setAspectRatio] = useState('16:9')
  const [isGenerating, setIsGenerating] = useState(false)
  const [videoUrl, setVideoUrl] = useState<string | null>(null)
  const [error, setError] = useState<string | null>(null)

  const { openInsufficientModal } = useCredits()

  const handleGenerate = useCallback(async () => {
    if (!prompt.trim()) {
      setError('请输入视频描述')
      return
    }

    setIsGenerating(true)
    setError(null)

    try {
      const response = await fetch('/api/video/generate', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        credentials: 'include',
        body: JSON.stringify({ model, prompt, imageUrl, aspectRatio }),
      })

      const data = await response.json()

      if (!data.success) {
        if (data.error === '积分不足') {
          openInsufficientModal()
        } else {
          setError(data.error)
        }
        return
      }

      // 开始轮询任务状态
      pollTaskStatus(data.taskId)
    } catch (err) {
      setError('视频生成失败')
    } finally {
      setIsGenerating(false)
    }
  }, [model, prompt, imageUrl, aspectRatio, openInsufficientModal])

  const pollTaskStatus = async (taskId: string) => {
    const poll = async () => {
      const response = await fetch(`/api/video/task/${taskId}`, {
        credentials: 'include',
      })
      const data = await response.json()

      if (data.status === 'completed' && data.videoUrl) {
        setVideoUrl(data.videoUrl)
        onVideoGenerated?.(data.videoUrl)
        return
      }

      if (data.status === 'failed') {
        setError('视频生成失败')
        return
      }

      // 继续轮询
      setTimeout(poll, 3000)
    }

    poll()
  }

  return (
    <div className="bg-white rounded-xl p-6 shadow-sm">
      <h3 className="font-sans-zh font-semibold text-lg mb-4">AI 视频生成</h3>

      <div className="mb-4">
        <label className="block text-sm font-medium text-gray-700 mb-2">生成模型</label>
        <div className="grid grid-cols-2 gap-2">
          {VIDEO_MODELS.map((m) => (
            <button
              key={m.id}
              onClick={() => setModel(m.id)}
              className={`p-3 rounded-lg border-2 transition-colors ${
                model === m.id
                  ? 'border-blue-500 bg-blue-50'
                  : 'border-gray-200 hover:border-gray-300'
              }`}
            >
              <div className="font-medium text-sm">{m.name}</div>
              <div className="text-xs text-gray-500">{m.cost} 积分/次</div>
            </button>
          ))}
        </div>
      </div>

      {/* 提示词 */}
      <div className="mb-4">
        <label className="block text-sm font-medium text-gray-700 mb-2">
          视频描述
        </label>
        <textarea
          value={prompt}
          onChange={(e) => setPrompt(e.target.value)}
          placeholder="描述视频内容，如：模特转身展示服装，背景为简约工作室..."

        <label className="block text-sm font-medium text-gray-700 mb-2">视频比例</label>

                  : 'border-gray-200 hover:border-gray-300'

            >
              {ratio.name}
            </button>
          ))}
        </div>
      </div>

      {/* 生成按钮 */}
      <button
        onClick={handleGenerate}
        disabled={isGenerating}
        className="w-full py-3 bg-blue-500 text-white rounded-lg font-medium hover:bg-blue-600 disabled:bg-gray-300 flex items-center justify-center gap-2"

          </>
```
