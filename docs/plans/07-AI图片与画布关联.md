# AI 生成图片与画布关联开发计划

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** 实现 AI 对话中生成的图片能够自动关联到画布组件，用户可以选择将生成的图片添加到画布或作为后续换装的输入。

**Architecture:** AI 生成的图片会先存储到临时区域，并记录生成元数据（提示词、模型、尺寸等）。用户可以通过点击或语音指令将图片插入画布，系统会自动创建对应的 Shape 并关联到 AI 组合组件。

**Tech Stack:** Next.js 16, React, Zustand, TypeScript

```ts
import { create } from 'zustand'

interface GeneratedImage {
  id: string
  imageUrl: string
  prompt: string
  model: string
  resolution: { width: number; height: number }
  createdAt: Date
  usedInCanvas: boolean
  canvasShapeId?: string
}

interface ImageGenerationState {
  generatedImages: GeneratedImage[]
  currentDraft: GeneratedImage | null
  addGeneratedImage: (image: GeneratedImage) => void
  markAsUsed: (imageId: string, canvasShapeId: string) => void
  clearDraft: () => void
  setDraft: (image: GeneratedImage) => void
}

export const useImageGenerationStore = create<ImageGenerationState>((set) => ({
  generatedImages: [],
  currentDraft: null,

  addGeneratedImage: (image) =>
    set((state) => ({
      generatedImages: [image, ...state.generatedImages],
      currentDraft: image,
    })),

  markAsUsed: (imageId, canvasShapeId) =>
    set((state) => ({
      generatedImages: state.generatedImages.map((img) =>
        img.id === imageId
          ? { ...img, usedInCanvas: true, canvasShapeId }
          : img
      ),
      currentDraft: null,
    })),

  clearDraft: () => set({ currentDraft: null }),

  setDraft: (image) => set({ currentDraft: image }),
}))
```

```ts
interface ChatResponse {
  message: string
  toolCalls?: Array<{
    tool: string
    args: Record<string, any>
  }>
  generatedImages?: Array<{
    url: string
    prompt: string
    model: string
    resolution: { width: number; height: number }
  }>
}

export async function sendChatMessage(
  message: string,
  projectId: string,
  options?: {
    model?: string
    context?: 'generate' | 'tryon' | 'edit'
  }
): Promise<ChatResponse> {
  const response = await fetch('/api/chat', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    credentials: 'include',
    body: JSON.stringify({ message, projectId, ...options }),
  })

  const data = await response.json()

  // 解析 AI 返回的消息，提取图片信息
  const generatedImages = parseGeneratedImages(data)

  return {
    message: data.message,
    toolCalls: data.toolCalls,
    generatedImages,
  }
}

function parseGeneratedImages(data: any): ChatResponse['generatedImages'] {
  if (!data.generatedImages) return []

  return data.generatedImages.map((img: any) => ({
    url: img.url,
    prompt: img.prompt || '',
    model: img.model || 'gemini-2.5-flash-image',
    resolution: img.resolution || { width: 1024, height: 1024 },
  }))
}
```

```tsx
'use client'

import React, { useState } from 'react'
import { Plus, X, Copy, Check, Loader2 } from 'lucide-react'
import { useImageGenerationStore } from '../stores/ImageGenerationStore'
import { useCanvasStore } from '@/app/canvas/store'

interface GeneratedImagePreviewProps {
  image: {
    id: string
    imageUrl: string
    prompt: string
    model: string
    resolution: { width: number; height: number }
  }
  onClose: () => void
}

export function GeneratedImagePreview({
  image,
  onClose,
}: GeneratedImagePreviewProps) {
  const [isAddingToCanvas, setIsAddingToCanvas] = useState(false)
  const [addedToCanvas, setAddedToCanvas] = useState(false)
  const { addShape } = useCanvasStore()
  const { markAsUsed } = useImageGenerationStore()

  const handleAddToCanvas = async () => {
    setIsAddingToCanvas(true)

    try {
      // 在画布上创建图片元素
      const newShape = addShape({
        type: 'image',
        x: 100,
        y: 100,
        width: Math.min(image.resolution.width / 2, 400),
        height: Math.min(image.resolution.height / 2, 400),
        imageUrl: image.imageUrl,
        imageName: `AI生成-${image.prompt.substring(0, 20)}`,
        imageWidth: image.resolution.width,
        imageHeight: image.resolution.height,
        aiGenerated: true,
        generationPrompt: image.prompt,
        generationModel: image.model,
      })

      // 标记为已使用
      markAsUsed(image.id, newShape.id)

      setAddedToCanvas(true)

      // 3秒后关闭预览
      setTimeout(() => {
        onClose()
      }, 1500)
    } catch (error) {
      console.error('Failed to add to canvas:', error)
    } finally {
      setIsAddingToCanvas(false)
    }
  }

  return (
    <div className="fixed bottom-4 right-4 z-50 w-80 bg-white rounded-xl shadow-2xl overflow-hidden">
      {/* 头部 */}
      <div className="flex items-center justify-between px-4 py-3 bg-gray-50 border-b">
        <span className="text-sm font-medium">AI 生成的图片</span>
        <button
          onClick={onClose}
          className="p-1 hover:bg-gray-200 rounded"
        >
          <X className="w-4 h-4" />
        </button>
      </div>

      {/* 图片预览 */}
      <div className="relative aspect-square bg-gray-100">
        <img
          src={image.imageUrl}
          alt={image.prompt}
          className="w-full h-full object-contain"
        />
      </div>

      {/* 图片信息 */}
      <div className="px-4 py-2 border-t text-xs text-gray-500">
        <div>模型：{image.model}</div>
        <div>尺寸：{image.resolution.width} × {image.resolution.height}</div>
      </div>

      {/* 操作按钮 */}
      <div className="p-4 pt-0 flex gap-2">
        <button
          onClick={handleAddToCanvas}
          disabled={isAddingToCanvas || addedToCanvas}
          className={`flex-1 py-2 rounded-lg text-sm font-medium flex items-center justify-center gap-2 ${
            addedToCanvas
              ? 'bg-green-500 text-white'
              : 'bg-blue-500 text-white hover:bg-blue-600 disabled:bg-gray-300'
          }`}
        >
          {isAddingToCanvas ? (
            <>
              <Loader2 className="w-4 h-4 animate-spin" />
              <span>添加中...</span>
            </>
          ) : addedToCanvas ? (
            <>
              <Check className="w-4 h-4" />
              <span>已添加</span>
            </>
          ) : (
            <>
              <Plus className="w-4 h-4" />
              <span>添加到画布</span>
            </>
          )}
        </button>
      </div>
    </div>
  )
}
```

```tsx
'use client'

import React, { useState } from 'react'
import { Clock, Plus, Trash2, Image as ImageIcon } from 'lucide-react'
import { useImageGenerationStore } from '../stores/ImageGenerationStore'

export function ImageHistoryPanel() {
  const { generatedImages, markAsUsed } = useImageGenerationStore()
  const [selectedImage, setSelectedImage] = useState<string | null>(null)

  if (generatedImages.length === 0) {
    return null
  }

  return (
    <div className="w-64 bg-white border-l h-full overflow-y-auto">
      <div className="p-4 border-b">
        <h3 className="font-medium flex items-center gap-2">
          <Clock className="w-4 h-4" />
          <span>生成历史</span>
        </h3>
      </div>

      <div className="p-2 space-y-2">
        {generatedImages.map((image) => (
          <div
            key={image.id}
            className={`relative group rounded-lg overflow-hidden cursor-pointer transition-all ${
              selectedImage === image.id ? 'ring-2 ring-blue-500' : ''
            }`}
            onClick={() => setSelectedImage(image.id)}
          >
            <img
              src={image.imageUrl}
              alt={image.prompt}
              className="w-full aspect-square object-cover"
            />

            {/* 悬停操作 */}
            <div className="absolute inset-0 bg-black/50 opacity-0 group-hover:opacity-100 transition-opacity flex items-center justify-center gap-2">
              {image.usedInCanvas ? (
                <span className="text-white text-xs">已使用</span>
              ) : (
                <button
                  onClick={(e) => {
                    e.stopPropagation()
                    // 添加到画布
                    markAsUsed(image.id, 'new-shape')
                  }}
                  className="p-2 bg-white rounded-full hover:bg-gray-100"
                >
                  <Plus className="w-4 h-4" />
                </button>
              )}
            </div>

            {/* 使用状态标记 */}
            {image.usedInCanvas && (
              <div className="absolute top-2 right-2 px-2 py-1 bg-green-500 text-white text-xs rounded-full">
                已使用
              </div>
            )}
          </div>
        ))}
      </div>
    </div>
  )
```

**Step 1: 创建插入到换装组件**

```tsx
'use client'

import React, { useState } from 'react'
import { X, Loader2, ChevronRight } from 'lucide-react'
import { useCanvasStore } from '@/app/canvas/store'
import { combinationRegistry } from '@/ai-combination/registry'

interface InsertToTryOnModalProps {
  image: {
    id: string
    imageUrl: string
    type: 'model' | 'clothing'
  }
  onClose: () => void
  onInsert: (shapeId: string, slotType: string) => void
}

export function InsertToTryOnModal({
  image,
  onClose,
  onInsert,
}: InsertToTryOnModalProps) {
  const [selectedType, setSelectedType] = useState<'model' | 'clothing'>(
    image.type
  )
  const { shapes } = useCanvasStore()

  // 找到画布上的 AI 组合组件
  const aiCombinationShapes = shapes.filter(
    (s) => s.type === 'ai-combination'
  )

  const handleInsert = (shapeId: string) => {
    const slotType = selectedType === 'model' ? 'model' : 'clothing'
    onInsert(shapeId, slotType)
    onClose()
  }

  return (
    <div className="fixed inset-0 z-50 bg-black/50 flex items-center justify-center">
      <div className="bg-white rounded-2xl w-full max-w-md p-6">
        <div className="flex items-center justify-between mb-4">
          <h3 className="font-semibold">插入到 AI 换装组件</h3>
          <button onClick={onClose} className="p-1 hover:bg-gray-100 rounded">
            <X className="w-5 h-5" />
          </button>
        </div>

        {/* 图片预览 */}
        <div className="flex gap-4 mb-6">
          <img
            src={image.imageUrl}
            alt="preview"
            className="w-24 h-24 object-cover rounded-lg"
          />
          <div className="flex-1">
            <p className="text-sm text-gray-600 mb-2">选择插入为：</p>
            <div className="flex gap-2">
              <button
                onClick={() => setSelectedType('model')}
                className={`px-4 py-2 rounded-lg text-sm ${
                  selectedType === 'model'
                    ? 'bg-blue-500 text-white'
                    : 'bg-gray-100'
                }`}
              >
                模特图
              </button>
              <button
                onClick={() => setSelectedType('clothing')}
                className={`px-4 py-2 rounded-lg text-sm ${
                  selectedType === 'clothing'
                    ? 'bg-blue-500 text-white'
                    : 'bg-gray-100'
                }`}
              >
                服装图
              </button>
            </div>
          </div>
        </div>

        {/* 选择目标 AI 组合组件 */}
        <div>
          <p className="text-sm text-gray-600 mb-2">选择目标 AI 换装组件：</p>
          {aiCombinationShapes.length === 0 ? (
            <p className="text-sm text-gray-400 py-4 text-center">
              画布上还没有 AI 换装组件，请先创建一个
            </p>
          ) : (
            <div className="space-y-2">
              {aiCombinationShapes.map((shape) => (
                <button
                  key={shape.id}
                  onClick={() => handleInsert(shape.id)}
                  className="w-full p-3 bg-gray-50 hover:bg-gray-100 rounded-lg text-left flex items-center justify-between"
                >
                  <span className="text-sm">
                    {shape.combinationTypeId || 'AI 换装'}
                  </span>
                  <ChevronRight className="w-4 h-4" />
                </button>
              ))}
            </div>
          )}
        </div>
      </div>
    </div>
  )
}
```
