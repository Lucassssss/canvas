**Goal:** 在 AI 对话界面中允许用户选择不同的模型、设置图片尺寸、比例和生成数量，并在后续的 AI 生成操作中使用这些设置。

```ts
import { create } from 'zustand'
import { persist } from 'zustand/middleware'

export interface ChatConfig {
  model: string
  imageModel: string
  resolution: '1K' | '2K' | '4K'
  aspectRatio: '1:1' | '3:4' | '4:3' | '9:16' | '16:9'
  count: number
}

interface ChatConfigState {
  config: ChatConfig
  updateConfig: (updates: Partial<ChatConfig>) => void
  resetConfig: () => void
}

const DEFAULT_CONFIG: ChatConfig = {
  model: 'claude-3-5-sonnet',
  imageModel: 'gemini-2.5-flash-image',
  resolution: '2K',
  aspectRatio: '1:1',
  count: 1,
}

export const useChatConfigStore = create<ChatConfigState>()(
  persist(
    (set) => ({
      config: DEFAULT_CONFIG,

      updateConfig: (updates) =>
        set((state) => ({
          config: { ...state.config, ...updates },
        })),

      resetConfig: () => set({ config: DEFAULT_CONFIG }),
    }),
    {
      name: 'joii-chat-config',
    }
  )
)
```

```tsx
'use client'

import React, { useState, useRef, useEffect } from 'react'
import { useChatConfigStore } from '../stores/ChatConfigStore'
import { ChatConfigPanel } from '../components/ChatConfigPanel'
import { ChatConfigBar } from '../components/ChatConfigBar'
import { ChatMessage } from '../components/ChatMessage'
import { ChatInput } from '../components/ChatInput'
import { GeneratedImagePreview } from '../components/GeneratedImagePreview'
import { useImageGenerationStore } from '../stores/ImageGenerationStore'

interface Message {
  id: string
  role: 'user' | 'assistant'
  content: string
  generatedImages?: any[]
}

export function ChatSection({ projectId }: { projectId: string }) {
  const [messages, setMessages] = useState<Message[]>([])
  const [isLoading, setIsLoading] = useState(false)
  const messagesEndRef = useRef<HTMLDivElement>(null)
  const { config } = useChatConfigStore()
  const { currentDraft, clearDraft } = useImageGenerationStore()

  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' })
  }, [messages])

  const handleSend = async (message: string, context: any) => {
    // 添加用户消息
    setMessages((prev) => [
      ...prev,
      { id: Date.now().toString(), role: 'user', content: message },
    ])

    setIsLoading(true)

    try {
      const response = await fetch('/api/chat', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        credentials: 'include',
        body: JSON.stringify({
          message,
          projectId,
          config: {
            model: config.model,
            imageModel: config.imageModel,
            resolution: config.resolution,
            aspectRatio: config.aspectRatio,
            count: config.count,
          },
          mentionContext: context.mentions,
        }),
      })

      const data = await response.json()

      setMessages((prev) => [
        ...prev,
        {
          id: (Date.now() + 1).toString(),
          role: 'assistant',
          content: data.message,
          generatedImages: data.generatedImages,
        },
      ])
    } catch (error) {
      setMessages((prev) => [
        ...prev,
        {
          id: (Date.now() + 1).toString(),
          role: 'assistant',
          content: '抱歉，发生了错误。请稍后重试。',
        },
      ])
    } finally {
      setIsLoading(false)
    }
  }

  return (
    <div className="flex flex-col h-full">
      {/* 配置栏 */}
      <ChatConfigBar />

      {/* 隐藏的配置按钮（用于触发面板） */}
      <button
        id="chat-config-btn"
        className="hidden"
        onClick={() => {}}
      />

      {/* 配置面板 */}
      <div className="absolute top-20 right-4 z-50">
        <ChatConfigPanel />
      </div>

      {/* 消息列表 */}
      <div className="flex-1 overflow-y-auto p-4 space-y-4">
        {messages.map((msg) => (
          <ChatMessage
            key={msg.id}
            message={msg}
            onImageClick={(img) => {
              useImageGenerationStore.getState().addGeneratedImage({
                id: Date.now().toString(),
                ...img,
                createdAt: new Date(),
                usedInCanvas: false,
              })
            }}
          />
        ))}
        {isLoading && (
          <div className="flex items-center gap-2 text-gray-500">
            <Loader2 className="w-4 h-4 animate-spin" />
            <span>AI 正在思考...</span>
          </div>
        )}
        <div ref={messagesEndRef} />
      </div>

      {/* 输入框 */}
      <div className="p-4 border-t">
        <ChatInput onSend={handleSend} />
      </div>

      {/* 图片预览 */}
      {currentDraft && (
        <GeneratedImagePreview
          image={currentDraft}
          onClose={clearDraft}
        />
      )}
    </div>
  )
}
```

```ts
interface ChatRequest {
  message: string
  projectId: string
  config?: {
    model?: string
    imageModel?: string
    resolution?: '1K' | '2K' | '4K'
    aspectRatio?: '1:1' | '3:4' | '4:3' | '9:16' | '16:9'
    count?: number
  }
  mentionContext?: {
    mentions: Array<{
      type: 'selected' | 'last' | 'all' | 'id'
      elementIds?: string[]
    }>
  }
}

router.post('/chat', async (req: Request, res: Response) => {
  const { message, projectId, config, mentionContext } = req.body as ChatRequest

  // 处理配置
  const imageConfig = {
    model: config?.imageModel || 'gemini-2.5-flash-image',
    resolution: config?.resolution || '2K',
    aspectRatio: config?.aspectRatio || '1:1',
    count: config?.count || 1,
  }

  // 将配置注入到消息中供 AI 使用
  const enhancedMessage = `${message}

[用户生成配置]
- 图片模型: ${imageConfig.model}
- 分辨率: ${imageConfig.resolution}
- 比例: ${imageConfig.aspectRatio}
- 数量: ${imageConfig.count}`

  const result = await agentService.chat(
    req.user!.id,
    projectId,
    enhancedMessage,
    { imageConfig }
  )

  res.json(result)
})
```
